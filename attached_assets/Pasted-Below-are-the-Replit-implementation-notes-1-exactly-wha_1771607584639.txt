Scoring patch: Items 8 and 10 (wall scoring)

Replace your current wall scorer with this logic (TypeScript-style; easy to translate).

type Tri = "YES" | "NO" | "NULL";
type Obs = "OBSERVED" | "NOT_OBSERVED";

function scoreWallFunctional(wall: any, label: "posterior"|"anterior"): {score: Tri, observability: Obs, evidence: any} {
  const completed = wall?.completed;
  const gaps = wall?.visible_gaps;              // true|false|"unknown"|undefined
  const gapConf = wall?.gap_confidence ?? "unknown";  // "high"|"low"|"unknown"
  const uniform = wall?.spacing_uniformity;     // "uniform"|"mixed"|"irregular"|"unknown"

  // If key fields are missing/unknown, be honest:
  if (completed === undefined || completed === "unknown") {
    return { score: "NULL", observability: "NOT_OBSERVED", evidence: { reason: `${label} wall completion not visible`, wall } };
  }
  if (uniform === undefined || uniform === "unknown") {
    return { score: "NULL", observability: "NOT_OBSERVED", evidence: { reason: `${label} wall spacing not visible`, wall } };
  }
  if (gaps === undefined || gaps === "unknown") {
    return { score: "NULL", observability: "NOT_OBSERVED", evidence: { reason: `${label} wall gaps not assessable`, wall } };
  }

  // Hard NO only for high-confidence failures:
  if (completed === false) {
    return { score: "NO", observability: "OBSERVED", evidence: { reason: `${label} wall not completed`, wall } };
  }

  if (gaps === true) {
    if (gapConf === "high") {
      return { score: "NO", observability: "OBSERVED", evidence: { reason: `${label} wall has high-confidence visible gaps/malapposition`, wall } };
    }
    // subtle/uncertain gaps -> NULL, not NO
    return { score: "NULL", observability: "NOT_OBSERVED", evidence: { reason: `${label} wall possible/subtle gaps (low confidence)`, wall } };
  }

  // Irregular spacing is only an automatic NO if it is paired with malapposition.
  // Without a malapposition signal, treat "irregular" as NULL unless you have a separate "leak-risk" field.
  if (uniform === "irregular") {
    return { score: "NULL", observability: "NOT_OBSERVED", evidence: { reason: `${label} wall spacing labeled irregular but no high-confidence malapposition evidence`, wall } };
  }

  // Otherwise YES:
  return { score: "YES", observability: "OBSERVED", evidence: { reason: `${label} wall completed, no visible gaps, spacing acceptable`, wall } };
}

Scoring patch: Economy (Item 12) without clip length

Stop requiring clip_minutes. Compute an event-weighted EI directly.

type EconomyType =
  "instrument_search" | "pause_reset" | "failed_pass_sequence" | "excess_regrasp_cluster" | "other";

const WEIGHTS: Record<EconomyType, number> = {
  instrument_search: 2,
  pause_reset: 2,
  failed_pass_sequence: 2,
  excess_regrasp_cluster: 1,
  other: 1
};

function computeEconomy(econ: any): {score: number | null, evidence: any} {
  const flow = econ?.flow_organization ?? "unknown";
  const events = Array.isArray(econ?.wasted_motion_events) ? econ.wasted_motion_events : [];

  // If there is truly no economy info, return NULL.
  const hasAnyInfo = flow !== "unknown" || events.length > 0;
  if (!hasAnyInfo) return { score: null, evidence: { reason: "No economy markers available", flow_organization: flow, wasted_motion_events: [] } };

  // Collapse micro-adjustments: if events include many tiny "needle_reload_adjustment" entries,
  // upstream should convert these into one "excess_regrasp_cluster" unless they caused failed_pass_sequence.
  // Here, just compute the EI.
  let EI = 0;
  for (const e of events) {
    const t = e.type as EconomyType;
    const c = Number(e.count_estimate ?? 0);
    if (!WEIGHTS[t] || c <= 0) continue;
    EI += WEIGHTS[t] * c;
  }

  // Default: cap at 4 to avoid “5 inflation” (matches your hawk/dove moderation goal)
  // If you want 5 enabled later, do it behind a flag and require EI===0.
  let score: number;
  if (EI <= 1) score = 4;
  else if (EI <= 3) score = 3;
  else if (EI <= 6) score = 2;
  else score = 1;

  return {
    score,
    evidence: { flow_organization: flow, wasted_motion_events: events, economy_index: EI }
  };
}

How to store it in the evidence-based output:

If score === null: item 12 score = "NULL", observability = "NOT_OBSERVED"

Else: item 12 score = score, observability = "OBSERVED"

Projection rule stays:

Proficiency derivation patch (core-domain NULL => insufficient evidence)

type Proficiency3 = "PROFICIENT" | "NOT_PROFICIENT" | "INSUFFICIENT_EVIDENCE";

function deriveProficiency(itemsById: Map<number, {score: Tri}>): {score: Proficiency3, evidence: any} {
  const coreIds = [2, 8, 10, 11];
  const missing: string[] = [];
  const red: string[] = [];

  for (const id of coreIds) {
    const s = itemsById.get(id)?.score;
    if (s === "NULL" || s === undefined) missing.push(`Item ${id}`);
    if (s === "NO") red.push(`Item ${id}`);
  }

  // Heel red-line only if it is an OBSERVED defect, not mere ambiguity.
  const item5 = itemsById.get(5)?.score;
  if (item5 === "NO") red.push("Item 5 (observed heel defect)");

  if (red.length > 0) return { score: "NOT_PROFICIENT", evidence: { red_lines_triggered: red, missing_core_domains: missing } };
  if (missing.length > 0) return { score: "INSUFFICIENT_EVIDENCE", evidence: { red_lines_triggered: [], missing_core_domains: missing } };
  return { score: "PROFICIENT", evidence: { red_lines_triggered: [], missing_core_domains: [] } };
}
